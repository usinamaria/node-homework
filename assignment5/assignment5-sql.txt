-- Task 1: Total price of each of the first 5 orders, ordered by order_id
SELECT orders.order_id, SUM(line_items.quantity * products.price) AS total_price
FROM orders
JOIN line_items ON orders.order_id = line_items.order_id
JOIN products ON line_items.product_id = products.product_id
GROUP BY orders.order_id
ORDER BY orders.order_id
LIMIT 5;

-- Task 2: Average total order price for each customer
SELECT customer_name, AVG(t.total_price) AS average_order_price
FROM customers c
JOIN (
  SELECT orders.order_id, orders.customer_id, SUM(line_items.quantity * products.price) AS total_price
  FROM orders
  JOIN line_items ON orders.order_id = line_items.order_id
  JOIN products ON line_items.product_id = products.product_id
  GROUP BY orders.order_id, orders.customer_id
) AS t ON c.customer_id = t.customer_id
GROUP BY c.customer_id, customer_name
ORDER BY customer_name;

-- Task 3: Creating a new order

-- Find the customer_id for Perez and Sons
SELECT customer_id FROM customers WHERE customer_name = 'Perez and Sons';

-- Find the employee_id for Miranda Harris
SELECT employee_id FROM employees WHERE first_name = 'Miranda' AND last_name = 'Harris';

-- Find the product_id for the 5 least expensive products
SELECT product_id FROM products ORDER BY price ASC LIMIT 5;

-- Start the transaction
BEGIN;

-- Insert the order, insert its 5 line items using the order_id from
-- RETURNING (no hardcoded order_id, no hardcoded product_ids), and
-- return the inserted line items as the result of this statement
WITH new_order AS (
  INSERT INTO orders (customer_id, employee_id, date)
  VALUES (
    (SELECT customer_id FROM customers WHERE customer_name = 'Perez and Sons'),
    (SELECT employee_id FROM employees WHERE first_name = 'Miranda' AND last_name = 'Harris'),
    '2024-01-01'
  )
  RETURNING order_id
),
cheapest_products AS (
  SELECT product_id
  FROM products
  ORDER BY price ASC
  LIMIT 5
)
INSERT INTO line_items (order_id, product_id, quantity)
SELECT new_order.order_id, cheapest_products.product_id, 10
FROM new_order, cheapest_products
RETURNING *;

-- Commit the transaction
COMMIT;

-- Task 4: Employees associated with more than 5 orders
SELECT employees.first_name, employees.last_name, COUNT(order_id) AS order_count
FROM employees
JOIN orders ON employees.employee_id = orders.employee_id
GROUP BY employees.employee_id, first_name, last_name
HAVING COUNT(order_id) > 5
ORDER BY last_name;
